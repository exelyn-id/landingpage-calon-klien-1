export function getWhatsAppUrl(message?: string) {
  const phone = "6282258009113";
  const defaultMessage = "Halo Zeyn Dental Care, saya lihat info dari Instagram @zeyndentalcare dan ingin tanya-tanya dulu.";
  const finalMessage = message ?? defaultMessage;
